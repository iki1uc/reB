function interpretVector(row) { return { id: row.id, tmp: row.tmp, sym: row.sym, color: row.color, axis: { x: (row.D - row.A), y: (row.W - row.S), z: (row.Y - row.C), rot: (row.E - row.Q) }, score: ( Number(row.W) + Number(row.A) + Number(row.D) + Number(row.S) + Number(row.Q) + Number(row.E) + Number(row.Y) + Number(row.C) ), vector: [ Number(row.W), Number(row.A), Number(row.D), Number(row.S), Number(row.Q), Number(row.E), Number(row.Y), Number(row.C) ] }; }
